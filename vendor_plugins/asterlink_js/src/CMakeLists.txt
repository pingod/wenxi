cmake_minimum_required(VERSION 3.14)
project(asterlink_js LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(asterlink_js SHARED bridge.c quickjs/quickjs.c quickjs/libregexp.c quickjs/libunicode.c quickjs/dtoa.c)
target_include_directories(asterlink_js PRIVATE quickjs)
set_target_properties(asterlink_js PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_definitions(asterlink_js PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(asterlink_js PRIVATE m)
endif()
if(ANDROID)
  target_link_options(asterlink_js PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
